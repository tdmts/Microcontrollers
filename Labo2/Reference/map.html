map()

Met map() kun je een waarde van het ene bereik omzetten naar een ander bereik. Dit is handig wanneer je bijvoorbeeld een sensorwaarde (0-1023) wilt omzetten naar een PWM-waarde (0-255), of wanneer je een percentage wilt berekenen.

Hoe werkt het?

long result = map(value, fromLow, fromHigh, toLow, toHigh);
  • value: de waarde die je wilt omzetten
  • fromLow: ondergrens van het invoerbereik
  • fromHigh: bovengrens van het invoerbereik
  • toLow: ondergrens van het uitvoerbereik
  • toHigh: bovengrens van het uitvoerbereik

Voorbeeld 1

Lineaire analogRead naar analogWrite

const int pinPotentiometer = A0;
const int pinLed = 3;

void setup()
{
  pinMode(pinLed, OUTPUT);
}

void loop()
{
  // Lees de potentiometer uit (0-1023)
  int valuePotentiometer = analogRead(pinPotentiometer);

  // Map van 0-1023 naar 0-255
  long brightness = map(valuePotentiometer, 0, 1023, 0, 255);

  // Stuur PWM signaal naar LED
  analogWrite(pinLed, brightness);
}

Voorbeeld 3

Dead zone potentiometer waarbij we de eerste 10% de LED volledig uitzetten en de laatste 10% de LED volledig aanzetten.

const int pinPotentiometer = A0;
const int pinLed = 3;

void setup()
{
  pinMode(pinLed, OUTPUT);
}

void loop()
{
  // Lees de potentiometer uit (0-1023)
  int valuePotentiometer = analogRead(pinPotentiometer);
  long brightness = 0;

  if(valuePotentiometer < 102)
  {
    brightness = 0;
  }
  else if(valuePotentiometer > 922)
  {
    brightness = 255;
  }
  else
  {
    // Map van 10%-90% naar 0%-100%
    brightness = map(valuePotentiometer, 102, 922, 0, 255);
  }

  // Stuur PWM signaal naar LED
  analogWrite(pinLed, brightness);
}
