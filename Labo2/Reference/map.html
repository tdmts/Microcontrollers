map()

Met map() kan je een waarde van het ene bereik omzetten naar een ander bereik. Dit is handig wanneer je bijvoorbeeld een sensorwaarde (0-1023) wil omzetten naar een PWM-waarde (0-255), of wanneer je een percentage wil berekenen.

Hoe werkt het?

long result = map(value, fromLow, fromHigh, toLow, toHigh);
  • value: de waarde die je wil omzetten
  • fromLow: ondergrens van het invoerbereik
  • fromHigh: bovengrens van het invoerbereik
  • toLow: ondergrens van het uitvoerbereik
  • toHigh: bovengrens van het uitvoerbereik

Voorbeeld 1

Lineaire analogRead naar analogWrite

const int potPin = A0;
const int ledPin = 3;

void setup()
{
  pinMode(ledPin, OUTPUT);
}

void loop()
{
  // Lees de potentiometer uit (0-1023)
  int potWaarde = analogRead(potPin);

  // Map van 0-1023 naar 0-255
  long helderheid = map(potWaarde, 0, 1023, 0, 255);

  // Stuur PWM signaal naar LED
  analogWrite(ledPin, helderheid);
}

Voorbeeld 3

Dead zone potentiometer waarbij we de eerste 10% de led volledig uitzetten en de laatste 10% de led volledig aanzetten.

const int potPin = A0;
const int ledPin = 3;

void setup()
{
  pinMode(ledPin, OUTPUT);
}

void loop()
{
  // Lees de potentiometer uit (0-1023)
  int potWaarde = analogRead(potPin);
  long helderheid = 0;

  if (potWaarde < 102)
  {
    helderheid = 0;
  }
  else if (potWaarde > 922)
  {
    helderheid = 255;
  }
  else
  {
    // Map van 10%-90% naar 0%-100%
    helderheid = map(potWaarde, 102, 922, 0, 255);
  }

  // Stuur PWM signaal naar LED
  analogWrite(ledPin, helderheid);
}
